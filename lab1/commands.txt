2. JENKINS INFRASTRUCTURE SETUP & CONTAINERIZATION
--------------------------------------------------------------------------------
# Create isolated Jenkins workspace directory
New-Item -ItemType Directory -Force -Path "lab1\jenkins"
New-Item -ItemType Directory -Force -Path "lab1\screenshots"
New-Item -ItemType Directory -Force -Path "lab1\reports"

# Pull official Jenkins LTS image
docker pull jenkins/jenkins:lts

# Build custom Jenkins container image equipped with Python3, Docker CLI, and Git
docker build -t lab1-jenkins-server:latest lab1/jenkins

# Create dedicated Docker network for Lab 1 CI/CD isolation
docker network create lab1-jenkins-net

# Launch Jenkins master container with Docker socket mount and port forwarding
docker run -d `
  --name lab1-jenkins `
  --network lab1-jenkins-net `
  -p 8080:8080 `
  -p 50000:50000 `
  -v /var/run/docker.sock:/var/run/docker.sock `
  -v lab1_jenkins_home:/var/jenkins_home `
  lab1-jenkins-server:latest

# Verify Jenkins master container health and logs
docker ps --filter "name=lab1-jenkins"
docker logs -n 50 lab1-jenkins
